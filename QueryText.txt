===================== Company Schedule — new table [24-09-2026] ======================

-- 2026-09-24-000001_AddCompanyScheduleTable
-- Not a ZKBioTime table — this app's own addition, so Company Schedule has a
-- real page/list to back it instead of silently fanning out into
-- att_departmentschedule rows. Modelled column-for-column on
-- att_departmentschedule (department_id -> company_id). Resolved by
-- AttendanceCalculator::shiftForEmployee() between Department and Group.

CREATE TABLE att_companyschedule (
  id            serial PRIMARY KEY,
  create_time   timestamp NULL,
  create_user   varchar(255) NULL,
  change_time   timestamp NULL,
  change_user   varchar(255) NULL,
  status        smallint NOT NULL DEFAULT 0,
  start_date    date NOT NULL,
  end_date      date NOT NULL,
  company_id    integer NOT NULL,
  shift_id      integer NOT NULL
);

CREATE INDEX att_companyschedule_company_id ON att_companyschedule (company_id);
CREATE INDEX att_companyschedule_shift_id   ON att_companyschedule (shift_id);
